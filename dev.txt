[ ] 1.3  Base Celery task pattern (features/tools/tasks.py)
         @app.task(queue='default')
         def process_tool(job_id):
             job = ToolJob.objects.get(id=job_id)
             job.status = 'processing'; job.save()
             try:
                 result_key = services.<tool>.run(job)
                 job.output_file = result_key
                 job.status = 'done'
             except Exception as e:
                 job.status = 'failed'
             job.save()

[ ] 1.4  HTMX polling pattern (partials/_job_status.jinja)
